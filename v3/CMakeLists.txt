cmake_minimum_required(VERSION 3.16)
project(rpp_dashboard VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick)

add_library(dashboard_backend STATIC dashboard.cpp dashboard.h sleepsettings.cpp sleepsettings.h)
target_link_libraries(dashboard_backend PUBLIC Qt6::Quick)
target_compile_options(dashboard_backend PRIVATE -Wall -Wextra -Wpedantic)

add_executable(rpp-dashboard main.cpp qml.qrc)
target_link_libraries(rpp-dashboard PRIVATE dashboard_backend)
install(TARGETS rpp-dashboard RUNTIME DESTINATION .)
install(PROGRAMS tablet/launch.sh DESTINATION .)

if(UNIX)
    add_library(appload_connection STATIC appload/connection.cpp appload/connection.h)
    target_link_libraries(appload_connection PUBLIC dashboard_backend)
    add_executable(dashboard-appload appload/backend.cpp)
    target_link_libraries(dashboard-appload PRIVATE appload_connection)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        install(TARGETS dashboard-appload RUNTIME DESTINATION .)
    endif()
    qt_add_binary_resources(appload-resources appload/application.qrc
        DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/resources.rcc")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/resources.rcc"
        appload/manifest.json DESTINATION appload)
    install(PROGRAMS appload/entry DESTINATION appload/backend)
endif()

option(BUILD_TESTING "Build desktop tests" OFF)
if(BUILD_TESTING)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test-dashboard tests/test_dashboard.cpp qml.qrc)
    target_link_libraries(test-dashboard PRIVATE dashboard_backend Qt6::Test)
    add_test(NAME dashboard COMMAND test-dashboard)
    set_tests_properties(dashboard PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    if(UNIX)
        add_executable(test-appload tests/test_appload.cpp)
        target_link_libraries(test-appload PRIVATE appload_connection Qt6::Test)
        add_dependencies(test-appload appload-resources)
        target_compile_definitions(test-appload PRIVATE
            APPLOAD_RESOURCE_PATH="${CMAKE_CURRENT_BINARY_DIR}/resources.rcc")
        add_test(NAME appload COMMAND test-appload)
        set_tests_properties(appload PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    endif()
endif()
